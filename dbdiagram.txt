// =====================
// USERS
// =====================

Table users {
  id int [pk, increment]
  firebase_uid varchar [unique, not null]
  email varchar
  created_at datetime
}

// =====================
// CAREERS / DEGREES
// =====================

Table degrees {
  id int [pk, increment]
  name varchar
}

Table user_study_plans {
  id int [pk, increment]
  user_id int
  study_plan_id int
  start_year int
  is_active boolean
}

// =====================
// STUDY PLANS
// =====================

Table study_plans {
  id int [pk, increment]
  degree_id int
  name varchar // "Plan 2008"
}

// =====================
// MODULES (niveles)
// =====================

Table modules {
  id int [pk, increment]
  study_plan_id int
  name varchar // "Primer Nivel"
  order_index int
}

// =====================
// SUBJECTS (materia abstracta)
// =====================

Table subjects {
  id int [pk, increment]
  code varchar
  name varchar
}

// =====================
// SUBJECT IN PLAN
// =====================

Table plan_subjects {
  id int [pk, increment]
  study_plan_id int
  subject_id int
  module_id int
  elective boolean
}

// =====================
// SUBJECT OFFERINGS (anual/cuatrimestral)
// =====================

Table subject_offerings {
  id int [pk, increment]
  subject_id int
  duration varchar // 'annual' | 'cuatrimestral'
  week_hours int
}

// =====================
// REQUIREMENTS (correlativas)
// =====================

Table requirements {
  id int [pk, increment]
  subject_id int
  required_subject_id int
  condition varchar // 'regularizada' | 'aprobada'
}

// =====================
// COURSES (comisiones reales)
// =====================

Table courses {
  id int [pk, increment]
  subject_offering_id int
  commission varchar
  year int
  professor varchar
}

// =====================
// ENROLLMENTS
// =====================

Table enrollments {
  id int [pk, increment]
  user_id int
  course_id int
  created_at datetime
}

// =====================
// EVALUATIONS (parcial/final)
// =====================

Table evaluations {
  id int [pk, increment]
  course_id int
  name varchar // "Parcial 1"
  type varchar // 'partial' | 'final'
  eval_order int
}

// =====================
// EVALUATION INSTANCES (fechas)
// =====================

Table evaluation_instances {
  id int [pk, increment]
  evaluation_id int
  date date
  instance_type varchar // 'regular' | 'recuperatorio'
  source varchar // 'user' | 'suggested' | 'official'
  created_by_user_id int
}

// =====================
// RESULTS
// =====================

Table evaluation_results {
  id int [pk, increment]
  user_id int
  evaluation_instance_id int
  grade decimal
  attended boolean
  created_at datetime
}

// =====================
// RELATIONSHIPS
// =====================

Ref: user_study_plans.user_id > users.id
Ref: user_study_plans.study_plan_id > study_plans.id

Ref: study_plans.degree_id > degrees.id
Ref: modules.study_plan_id > study_plans.id

Ref: plan_subjects.study_plan_id > study_plans.id
Ref: plan_subjects.subject_id > subjects.id
Ref: plan_subjects.module_id > modules.id

Ref: subject_offerings.subject_id > subjects.id

Ref: requirements.subject_id > subjects.id
Ref: requirements.required_subject_id > subjects.id

Ref: courses.subject_offering_id > subject_offerings.id

Ref: enrollments.user_id > users.id
Ref: enrollments.course_id > courses.id

Ref: evaluations.course_id > courses.id

Ref: evaluation_instances.evaluation_id > evaluations.id
Ref: evaluation_instances.created_by_user_id > users.id

Ref: evaluation_results.user_id > users.id
Ref: evaluation_results.evaluation_instance_id > evaluation_instances.id
